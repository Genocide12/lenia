    73→    73→/** Rotate a creature pattern by 90° increments (Lenia is rotation-equivariant). */
    74→    74→function rotateCells(cells: number[], w: number, h: number, quarterTurns: number): { cells: number[]; w: number; h: number } {
    75→    75→  let cur = cells.slice();
    76→    76→  let cw = w;
    77→    77→  let ch = h;
    78→    78→  for (let t = 0; t < ((quarterTurns % 4) + 4) % 4; t++) {
    79→    79→    const next = new Array(cw * ch);
    80→    80→    for (let y = 0; y < ch; y++) {
    81→    81→      for (let x = 0; x < cw; x++) {
    82→    82→        next[x * ch + (ch - 1 - y)] = cur[y * cw + x];
    83→    83→      }
    84→    84→    }
    85→    85→    cur = next;
    86→    86→    const tmp = cw;
    87→    87→    cw = ch;
    88→    88→    ch = tmp;
    89→    89→  }
    90→    90→  return { cells: cur, w: cw, h: ch };
    91→    91→}
